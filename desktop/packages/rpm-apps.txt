# Apps from the vendors' own repos (added in stage 2) and Fedora.

[Google Chrome]
google-chrome-stable    # Chrome browser (Google's repo)

[VS Code]
code                    # Visual Studio Code (Microsoft's repo)

[ChatGPT]
chatgpt                 # ChatGPT desktop, with Codex (OpenAI's repo)

[Google Cloud CLI]
google-cloud-cli                          # gcloud
google-cloud-cli-gke-gcloud-auth-plugin   # sign-in plugin for GKE clusters
kubectl                                   # Kubernetes CLI
google-cloud-cli-skaffold                 # build/deploy loop for Kubernetes apps

[Tailscale]
tailscale               # private network between your machines (Tailscale's repo)

[remote desktop]
remmina                     # remote desktop client
remmina-plugins-rdp         # RDP: Windows, and KDE's KRDP
remmina-plugins-vnc         # VNC
remmina-plugins-secret      # saves passwords in KWallet (Secret Service)
remmina-plugins-kwallet?    # direct KWallet plugin (not in every release)

[virtual machines]
@virtualization         # QEMU/KVM, libvirt and virt-manager
swtpm                   # virtual TPM chip (Windows 11 needs one)
swtpm-tools             # swtpm setup tools
edk2-ovmf               # UEFI firmware for VMs (Windows, and Claude Desktop's Cowork VM)
virtio-win              # Windows VM drivers ISO (virtio-win repo)

[Windows USB installer (WoeUSB-ng)]
python3-wxpython4           # GUI toolkit WoeUSB-ng uses
python3-termcolor           # WoeUSB-ng's coloured output
python3-pip                 # installs WoeUSB-ng from PyPI (Fedora doesn't package it)
parted                      # partitions the USB stick
dosfstools                  # FAT32 (mkfs.fat)
ntfs-3g                     # NTFS driver
ntfsprogs                   # mkntfs: NTFS for Windows 11's install.wim (over FAT32's 4 GB limit)
grub2-tools                 # grub2-install: makes the stick boot on legacy BIOS PCs
grub2-pc-modules            # GRUB's legacy BIOS (i386-pc) files

[Google Drive mount]
rclone                  # mounts Google Drive at ~/GoogleDrive
fuse3                   # lets rclone mount it as a normal folder

[Claude Desktop]
claude-desktop-unofficial   # Claude Desktop: Anthropic's official app, repackaged as an RPM by the claude-desktop-debian project (repo key fingerprint-checked)

[Cowork VM]
qemu-system-x86             # QEMU: runs Cowork's workspace virtual machine
virtiofsd                   # shares folders between Cowork's VM and this PC
